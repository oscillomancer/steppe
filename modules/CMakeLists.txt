# --- the stock modules ------------------------------------------------------------------
# Every stock/<name>.cpp is a module: its struct (the one deriving from
# Module) is found in the file, so adding a module is adding its file. Each
# is wrapped as C++ at build time (cmake/stock_module.cmake).
file(GLOB _stock_scripts CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/stock/*.cpp)
list(SORT _stock_scripts)
set(_stock_generated)
set(_stock_names)
foreach(_script IN LISTS _stock_scripts)
  get_filename_component(NAME ${_script} NAME_WE)
  if(NOT NAME MATCHES "^[a-z][a-z0-9_]*$")
    message(FATAL_ERROR "modules/stock/${NAME}.cpp: a stock module's file name is lower case "
                        "letters, digits and _")
  endif()
  set(_out ${CMAKE_CURRENT_BINARY_DIR}/stock/${NAME}.cpp)
  add_custom_command(
    OUTPUT ${_out}
    COMMAND ${CMAKE_COMMAND} -DNAME=${NAME} -DSCRIPT=${_script}
            -DTEMPLATE=${CMAKE_CURRENT_SOURCE_DIR}/cmake/stock_module.cpp.in -DOUT=${_out}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/stock_module.cmake
    DEPENDS ${_script} cmake/stock_module.cpp.in cmake/stock_module.cmake
    COMMENT "Stock module ${NAME}"
    VERBATIM)
  list(APPEND _stock_generated ${_out})
  list(APPEND _stock_names ${NAME})
endforeach()

set(_decls "")
set(_entries "")
foreach(_name IN LISTS _stock_names)
  string(APPEND _decls "const StockModule& module_${_name}();\n")
  string(APPEND _entries "        &stock::module_${_name}(),\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/stock/stock_list.cpp CONTENT
"// SPDX-License-Identifier: MIT
// Generated by modules/CMakeLists.txt: every stock module.
#include \"supple/rack/stock.h\"

namespace supple::rack {
namespace stock {
${_decls}}  // namespace stock

const std::vector<const StockModule*>& stockModules() {
    static const std::vector<const StockModule*> all = {
${_entries}    };
    return all;
}

}  // namespace supple::rack
" @ONLY)

# --- the factory rack patches --------------------------------------------------------------
# Every patches/<name>.json is a rack preset shipped with the app (rack.md
# §7.1), compiled in as bytes so no install path is needed.
file(GLOB _patch_files CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/patches/*.json)
list(SORT _patch_files)
# Read at configure time: an edited patch configures again.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_patch_files})
set(_patch_data "")
set(_patch_entries "")
set(_patch_index 0)
foreach(_patch IN LISTS _patch_files)
  get_filename_component(_patch_name ${_patch} NAME_WE)
  if(NOT _patch_name MATCHES "^[a-z0-9]+(-[a-z0-9]+)*$")
    message(FATAL_ERROR "modules/patches/${_patch_name}.json: a factory patch's file name is "
                        "lower case words joined by hyphens")
  endif()
  file(READ ${_patch} _hex HEX)
  string(LENGTH "${_hex}" _hex_length)
  math(EXPR _bytes "${_hex_length} / 2")
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _hex "${_hex}")
  string(APPEND _patch_data "const unsigned char patch_${_patch_index}[] = {${_hex}0x00};\n")
  string(APPEND _patch_entries
         "        {\"${_patch_name}\", {reinterpret_cast<const char*>(patch_${_patch_index}), ${_bytes}}},\n")
  math(EXPR _patch_index "${_patch_index} + 1")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/stock/patch_list.cpp CONTENT
"// SPDX-License-Identifier: MIT
// Generated by modules/CMakeLists.txt: every factory rack patch.
#include \"supple/rack/stock.h\"

namespace supple::rack {
namespace {
${_patch_data}}  // namespace

const std::vector<FactoryPatch>& factoryPatches() {
    static const std::vector<FactoryPatch> all = {
${_patch_entries}    };
    return all;
}

}  // namespace supple::rack
" @ONLY)

add_library(rack-stock STATIC ${_stock_generated} ${CMAKE_CURRENT_BINARY_DIR}/stock/stock_list.cpp
                              ${CMAKE_CURRENT_BINARY_DIR}/stock/patch_list.cpp)
target_include_directories(rack-stock PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/rack/include)
target_link_libraries(rack-stock PUBLIC supple::script-runtime)
supple_target_defaults(rack-stock)
# As a source export builds a script: no contraction, signed overflow wraps,
# so the C++ build plays the same bits as the JIT (ADR-0032).
supple_compile_options(rack-stock PRIVATE -ffp-contract=off -fwrapv)
set_target_properties(rack-stock PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(supple::rack-stock ALIAS rack-stock)
